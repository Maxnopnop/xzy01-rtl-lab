module TopModule (
    input  logic       clk,
    input  logic       reset,
    input  logic [2:0] s,
    output logic       fr2,
    output logic       fr1,
    output logic       fr0,
    output logic       dfr
);

    logic [2:0] level_state;
    logic       rising;

    always_ff @(posedge clk) begin
        if (reset) begin
            level_state <= 3'b000;
            rising      <= 1'b1;
        end else begin
            if (s != level_state) begin
                rising <= (s > level_state);
            end
            level_state <= s;
        end
    end

    always_comb begin
        fr2 = 1'b0;
        fr1 = 1'b0;
        fr0 = 1'b0;
        dfr = 1'b0;

        case (level_state)
            3'b000: begin
                fr2 = 1'b1;
                fr1 = 1'b1;
                fr0 = 1'b1;
                dfr = 1'b1;
            end
            3'b001: begin
                fr0 = 1'b1;
                dfr = rising;
            end
            3'b011: begin
                fr1 = 1'b1;
                fr0 = 1'b1;
                dfr = rising;
            end
            3'b111: begin
                fr2 = 1'b0;
                fr1 = 1'b0;
                fr0 = 1'b0;
                dfr = 1'b0;
            end
            default: begin
                fr2 = 1'b1;
                fr1 = 1'b1;
                fr0 = 1'b1;
                dfr = 1'b1;
            end
        endcase
    end

endmodule
