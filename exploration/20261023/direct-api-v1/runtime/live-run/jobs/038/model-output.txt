module top (
    input  logic x,
    input  logic y,
    output logic z
);
    logic a_out;
    logic b_out;
    logic or_out;
    logic and_out;

    assign a_out = (x ^ y) & x;
    assign b_out = ~(x ^ y);
    assign or_out = a_out | b_out;
    assign and_out = a_out & b_out;
    assign z = or_out ^ and_out;
endmodule
