module TopModule (
    input  logic       clk,
    input  logic       reset,
    input  logic [2:0] s,
    output logic       fr2,
    output logic       fr1,
    output logic       fr0,
    output logic       dfr
);

    logic [1:0] level;
    logic [1:0] sensed_level;
    logic       direction_up;

    always_comb begin
        if (!s[0])
            sensed_level = 2'd0;
        else if (!s[1])
            sensed_level = 2'd1;
        else if (!s[2])
            sensed_level = 2'd2;
        else
            sensed_level = 2'd3;
    end

    always_ff @(posedge clk) begin
        if (reset) begin
            level        <= 2'd0;
            direction_up <= 1'b1;
        end else begin
            level <= sensed_level;

            if (sensed_level > level)
                direction_up <= 1'b1;
            else if (sensed_level < level)
                direction_up <= 1'b0;
        end
    end

    always_comb begin
        fr0 = 1'b0;
        fr1 = 1'b0;
        fr2 = 1'b0;
        dfr = 1'b0;

        case (level)
            2'd0: begin
                fr0 = 1'b1;
                fr1 = 1'b1;
                fr2 = 1'b1;
                dfr = 1'b1;
            end
            2'd1: begin
                fr0 = 1'b1;
                fr1 = 1'b1;
                dfr = direction_up;
            end
            2'd2: begin
                fr0 = 1'b1;
                dfr = direction_up;
            end
            default: begin
                fr0 = 1'b0;
                fr1 = 1'b0;
                fr2 = 1'b0;
                dfr = 1'b0;
            end
        endcase
    end

endmodule
