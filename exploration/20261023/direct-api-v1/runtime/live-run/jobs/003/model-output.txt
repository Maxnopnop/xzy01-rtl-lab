module TopModule (
    input  logic       clk,
    input  logic       reset,
    input  logic [2:0] s,
    output logic       fr2,
    output logic       fr1,
    output logic       fr0,
    output logic       dfr
);

    logic [1:0] level_reg;
    logic [1:0] level_next;
    logic       valid_level;
    logic       direction_up;

    always_comb begin
        level_next = level_reg;
        valid_level = 1'b1;

        case (s)
            3'b000: level_next = 2'd0;
            3'b001: level_next = 2'd1;
            3'b011: level_next = 2'd2;
            3'b111: level_next = 2'd3;
            default: valid_level = 1'b0;
        endcase
    end

    always_ff @(posedge clk) begin
        if (reset) begin
            level_reg    <= 2'd0;
            direction_up <= 1'b1;
        end else if (valid_level) begin
            if (level_next != level_reg) begin
                direction_up <= (level_next > level_reg);
            end
            level_reg <= level_next;
        end
    end

    always_comb begin
        fr2 = 1'b0;
        fr1 = 1'b0;
        fr0 = 1'b0;

        case (level_reg)
            2'd0: begin
                fr2 = 1'b1;
                fr1 = 1'b1;
                fr0 = 1'b1;
            end
            2'd1: begin
                fr1 = 1'b1;
                fr0 = 1'b1;
            end
            2'd2: begin
                fr0 = 1'b1;
            end
            default: begin
                fr2 = 1'b0;
                fr1 = 1'b0;
                fr0 = 1'b0;
            end
        endcase
    end

    assign dfr = direction_up;

endmodule
