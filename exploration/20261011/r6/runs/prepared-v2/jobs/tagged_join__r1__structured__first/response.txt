```verilog
module tagged_join (
    input  wire       clk,
    input  wire       rst_n,
    input  wire       clear,
    input  wire       a_valid,
    input  wire [1:0] a_tag,
    input  wire [1:0] a_data,
    input  wire       b_valid,
    input  wire [1:0] b_tag,
    input  wire [1:0] b_data,
    input  wire       take,
    output reg        a_full,
    output reg        b_full,
    output reg        out_valid,
    output reg  [1:0] out_tag,
    output reg  [3:0] out_data
);

    reg [1:0] a_tag_hold;
    reg [1:0] a_data_hold;
    reg [1:0] b_tag_hold;
    reg [1:0] b_data_hold;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            a_full      <= 1'b0;
            a_tag_hold  <= 2'b00;
            a_data_hold <= 2'b00;
            b_full      <= 1'b0;
            b_tag_hold  <= 2'b00;
            b_data_hold <= 2'b00;
            out_valid   <= 1'b0;
            out_tag     <= 2'b00;
            out_data    <= 4'b0000;
        end else if (clear) begin
            a_full      <= 1'b0;
            a_tag_hold  <= 2'b00;
            a_data_hold <= 2'b00;
            b_full      <= 1'b0;
            b_tag_hold  <= 2'b00;
            b_data_hold <= 2'b00;
            out_valid   <= 1'b0;
            out_tag     <= 2'b00;
            out_data    <= 4'b0000;
        end else begin
            if (!a_full) begin
                if (a_valid) begin
                    a_full      <= 1'b1;
                    a_tag_hold  <= a_tag;
                    a_data_hold <= a_data;
                end
            end else if (b_full) begin
                if (a_tag_hold < b_tag_hold)
                    a_full <= 1'b0;
                else if ((a_tag_hold == b_tag_hold) &&
                         (!out_valid || take))
                    a_full <= 1'b0;
            end

            if (!b_full) begin
                if (b_valid) begin
                    b_full      <= 1'b1;
                    b_tag_hold  <= b_tag;
                    b_data_hold <= b_data;
                end
            end else if (a_full) begin
                if (b_tag_hold < a_tag_hold)
                    b_full <= 1'b0;
                else if ((a_tag_hold == b_tag_hold) &&
                         (!out_valid || take))
                    b_full <= 1'b0;
            end

            if (a_full && b_full &&
                (a_tag_hold == b_tag_hold) &&
                (!out_valid || take)) begin
                out_valid <= 1'b1;
                out_tag   <= a_tag_hold;
                out_data  <= {a_data_hold, b_data_hold};
            end else if (take) begin
                out_valid <= 1'b0;
            end
        end
    end

endmodule
```
