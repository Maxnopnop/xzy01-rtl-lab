module rotating_owner (
    input  logic clk,
    input  logic rst_n,
    input  logic req0,
    input  logic req1,
    input  logic retire,
    output logic grant0,
    output logic grant1
);

    logic prefer1;

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            grant0  <= 1'b0;
            grant1  <= 1'b0;
            prefer1 <= 1'b0;
        end else if (grant0 && retire) begin
            prefer1 <= 1'b1;
            grant0  <= req0 && !req1;
            grant1  <= req1;
        end else if (grant1 && retire) begin
            prefer1 <= 1'b0;
            grant0  <= req0;
            grant1  <= req1 && !req0;
        end else if (!grant0 && !grant1) begin
            if (req0 && req1) begin
                grant0 <= !prefer1;
                grant1 <= prefer1;
            end else begin
                grant0 <= req0;
                grant1 <= req1;
            end
        end
    end

endmodule
