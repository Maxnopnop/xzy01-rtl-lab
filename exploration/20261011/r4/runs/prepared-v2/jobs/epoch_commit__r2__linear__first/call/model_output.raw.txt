```systemverilog
module epoch_commit (
    input  logic       clk,
    input  logic       rst_n,
    input  logic       write,
    input  logic       slot,
    input  logic [1:0] data,
    input  logic       commit,
    input  logic       abort_tx,
    input  logic       read_slot,
    output logic [1:0] visible,
    output logic [1:0] dirty,
    output logic       committed
);

    logic [1:0] live0, live1;
    logic [1:0] staged0, staged1;
    logic [1:0] dirty_bits;
    logic       committed_reg;

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            live0         <= 2'b00;
            live1         <= 2'b00;
            staged0       <= 2'b00;
            staged1       <= 2'b00;
            dirty_bits    <= 2'b00;
            committed_reg <= 1'b0;
        end else if (abort_tx) begin
            staged0       <= live0;
            staged1       <= live1;
            dirty_bits    <= 2'b00;
            committed_reg <= 1'b0;
        end else begin
            committed_reg <= 1'b0;

            if (write) begin
                if (slot)
                    staged1 <= data;
                else
                    staged0 <= data;
            end

            if (commit) begin
                if (dirty_bits[0] || (write && !slot))
                    live0 <= (write && !slot) ? data : staged0;

                if (dirty_bits[1] || (write && slot))
                    live1 <= (write && slot) ? data : staged1;

                dirty_bits    <= 2'b00;
                committed_reg <= 1'b1;
            end else if (write) begin
                if (slot)
                    dirty_bits <= dirty_bits | 2'b10;
                else
                    dirty_bits <= dirty_bits | 2'b01;
            end
        end
    end

    always_comb begin
        visible   = !rst_n ? 2'b00 : (read_slot ? live1 : live0);
        dirty     = dirty_bits;
        committed = committed_reg;
    end

endmodule
```
