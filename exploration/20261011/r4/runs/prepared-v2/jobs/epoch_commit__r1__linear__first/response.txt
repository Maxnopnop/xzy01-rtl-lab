module epoch_commit (
    input  logic       clk,
    input  logic       rst_n,
    input  logic       write,
    input  logic       slot,
    input  logic [1:0] data,
    input  logic       commit,
    input  logic       abort_tx,
    input  logic       read_slot,
    output logic [1:0] visible,
    output logic [1:0] dirty,
    output logic       committed
);

    logic [1:0] live0, live1;
    logic [1:0] staged0, staged1;
    logic [1:0] dirty_r;
    logic       committed_r;

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            live0       <= 2'b00;
            live1       <= 2'b00;
            staged0     <= 2'b00;
            staged1     <= 2'b00;
            dirty_r     <= 2'b00;
            committed_r <= 1'b0;
        end else if (abort_tx) begin
            staged0     <= live0;
            staged1     <= live1;
            dirty_r     <= 2'b00;
            committed_r <= 1'b0;
        end else begin
            committed_r <= commit;

            if (commit) begin
                if (dirty_r[0] || (write && !slot))
                    live0 <= (write && !slot) ? data : staged0;
                if (dirty_r[1] || (write && slot))
                    live1 <= (write && slot) ? data : staged1;

                if (write && !slot)
                    staged0 <= data;
                if (write && slot)
                    staged1 <= data;

                dirty_r <= 2'b00;
            end else if (write) begin
                if (!slot) begin
                    staged0   <= data;
                    dirty_r[0] <= 1'b1;
                end else begin
                    staged1   <= data;
                    dirty_r[1] <= 1'b1;
                end
            end
        end
    end

    always_comb begin
        visible   = !rst_n ? 2'b00 : (read_slot ? live1 : live0);
        dirty     = dirty_r;
        committed = committed_r;
    end

endmodule
