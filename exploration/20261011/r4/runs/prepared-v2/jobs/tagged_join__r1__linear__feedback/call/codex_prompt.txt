    input  logic       btag,
    input  logic [1:0] bdata,
    input  logic       flush,
    output logic       pending_a,
    output logic       pending_b,
    output logic       joined,
    output logic [1:0] result
);

    logic       a_valid;
    logic       a_tag;
    logic [1:0] a_data;
    logic       b_valid;
    logic       b_tag;
    logic [1:0] b_data;

    logic       updated_a_tag;
    logic [1:0] updated_a_data;

    logic       updated_b_tag;
    logic [1:0] updated_b_data;

    always_comb begin
        updated_a_valid = av ? 1'b1 : a_valid;
        updated_a_tag   = av ? atag : a_tag;

            result  <= 2'b00;
