module refund_credit (
    input  logic       clk,
    input  logic       rst_n,
    input  logic       submit,
    input  logic [1:0] refund,
    input  logic [1:0] cost,
    output logic [2:0] credit,
    output logic       accepted,
    output logic       rejected,
    output logic       can_pay
);

    logic [2:0] credit_q;
    logic       accepted_q;
    logic       rejected_q;
    logic [3:0] credit_sum;
    logic [3:0] funded;

    always_comb begin
        credit_sum = {1'b0, credit_q} + {2'b00, refund};
        funded = (credit_sum >= 4'd5) ? 4'd5 : credit_sum;

        credit   = credit_q;
        accepted = accepted_q;
        rejected = rejected_q;
        can_pay  = rst_n &&
                   (cost != 2'd0) &&
                   (funded >= {2'b00, cost});
    end

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            credit_q   <= 3'd0;
            accepted_q <= 1'b0;
            rejected_q <= 1'b0;
        end else begin
            credit_q   <= funded[2:0];
            accepted_q <= 1'b0;
            rejected_q <= 1'b0;

            if (submit) begin
                if (cost == 2'd0) begin
                    rejected_q <= 1'b1;
                end else if (funded >= {2'b00, cost}) begin
                    credit_q   <= funded[2:0] - {1'b0, cost};
                    accepted_q <= 1'b1;
                end else begin
                    rejected_q <= 1'b1;
                end
            end
        end
    end

endmodule
